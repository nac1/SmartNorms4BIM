Polygon_2 create_polygon(std::vector<vertex> points)
{
    Polygon_2 poly;
    for (unsigned i = 0; i < points.size(); i++)
        poly.push_back(Point(points[i].x, points[i].y));

    return poly;
}

vector<line> create_lines(vector<vertex> points)
{
    vector<line> my_line;
    cout << "Create lines" << endl;
    for (unsigned i = 0; i < points.size(); i++)
    {
        line l;
        int aux = i + 1;
        if (aux == points.size())
        {
            //std::cout <<i<< "Create lines" << endl;
            l.p1 = Kernel::Point_2(points[i].x, points[i].y);
            l.p2 = Kernel::Point_2(points[0].x, points[0].y);
            l.seg = Segment_2(l.p1, l.p2);
            my_line.push_back(l);
        }
        else {
            l.p1 = Kernel::Point_2(points[i].x, points[i].y);
            l.p2 = Kernel::Point_2(points[i + 1].x, points[i + 1].y);
            l.seg = Segment_2(l.p1, l.p2);
            my_line.push_back(l);
        }
    }

    return my_line;
}

template<class K, class C>
offset_lines_min_max_xy see_polygon(CGAL::Polygon_2<K, C> const& poly)
{
    typedef CGAL::Polygon_2<K, C> Polygon;

    cout << "Polygon with " << poly.size() << " vertices" << endl;

    double maxAuxX = -1000, maxAuxY = -1000, minAuxX = 1000, minAuxY = 1000;
    vector<vertex> arrayXY;
    vector<vertex> arrayVertex;

    for (typename Polygon::Vertex_const_iterator vi = poly.vertices_begin(); vi != poly.vertices_end(); ++vi)
    {
        // print_point(*vi); cout << endl;
        Point p = get_pointOffset(*vi);
       
        cout << endl;
        if (maxAuxX < p.x())
            maxAuxX = p.x();

        if (maxAuxY < p.y())
            maxAuxY = p.y();

        if (minAuxX > p.x())
            minAuxX = p.x();

        if (minAuxY > p.y())
            minAuxY = p.y();

        vertex p1;
        p1.x = p.x();
        p1.y = p.y();
        arrayVertex.push_back(p1);// array of points from offset 
    }
    vertex p;
    cout << "x,y max" << maxAuxX << " ," << maxAuxY << endl;
    p.x = maxAuxX;
    p.y = maxAuxY;
    arrayXY.push_back(p); //position 0 max

    cout << "x,y min" << minAuxX << " ," << minAuxY << endl;
    p.x = minAuxX;
    p.y = minAuxY; //position 1 min
    arrayXY.push_back(p);

    vector<line> lines = create_lines(arrayVertex); //lines of offset

    offset_lines_min_max_xy offset_xy;
    offset_xy.min_max_xy = arrayXY;
    offset_xy.lines = lines;
    offset_xy.empty = false;

    return   offset_xy;
}


template<class K, class C>
offset_lines_min_max_xy get_Max_Min_polygons(std::vector< boost::shared_ptr< CGAL::Polygon_2<K, C> > > const& polies)
{
    typedef std::vector< boost::shared_ptr< CGAL::Polygon_2<K, C> > > PolygonVector;
    offset_lines_min_max_xy instance;

    cout << "Polygon list with " << polies.size() << " polygons" << endl;
    if (polies.size() == 1)
    {
        for (typename PolygonVector::const_iterator pi = polies.begin(); pi != polies.end(); ++pi)
            return see_polygon(**pi);
    }
    else
        return instance;
}

vector<Point> create_2DPoints( vector<vertex> array)
{
    vector < Point> vector;
    cout << "Create polygon 2d for centroid" << endl;

    for (unsigned i = 0; i < array.size(); i++)
        vector.push_back(Point(array[i].x, array[i].y));
    
    return vector;
}
